\mondai{90}{
次の関数の逆ラプラス変換を求めよ．
(1) $\dfrac{1}{s^2+4s+4}$ \quad
(2) $\dfrac{1}{s^2-4s+3}$ \quad
(3) $\dfrac{1}{s^2-6s+10}$
\hfill {\scriptsize [教p.59(問17)]}
}

\kotae{
(1) 平方完成：$s^2+4s+4=(s+2)^2$

$\mathcal{L}^{-1}\!\left[\dfrac{1}{(s+2)^2}\right] = te^{-2t}$

(2) 因数分解：$s^2-4s+3=(s-1)(s-3)$

$\dfrac{1}{(s-1)(s-3)} = \dfrac{A}{s-1}+\dfrac{B}{s-3}$ とおく

通分：$1=A(s-3)+B(s-1)$

$s=1$：$1=A(-2) \to A=-\dfrac{1}{2}$，\quad $s=3$：$1=B(2) \to B=\dfrac{1}{2}$

$\mathcal{L}^{-1} = -\dfrac{1}{2}e^t+\dfrac{1}{2}e^{3t} = \dfrac{1}{2}(e^{3t}-e^t)$

(3) 平方完成：$s^2-6s+10=(s-3)^2+1$

$\mathcal{L}^{-1}\!\left[\dfrac{1}{(s-3)^2+1}\right] = e^{3t}\sin t$
}

\mondai{91}{
次の関数の逆ラプラス変換を求めよ．
(1) $\dfrac{s+1}{s^2+6s+9}$ \quad
(2) $\dfrac{s}{s^2-4s+13}$
\hfill {\scriptsize [教p.59(問18)]}
}

\kotae{
(1) 平方完成：$s^2+6s+9=(s+3)^2$

$\dfrac{s+1}{(s+3)^2} = \dfrac{A}{s+3}+\dfrac{B}{(s+3)^2}$ とおく

通分：$s+1=A(s+3)+B$

$s=-3$：$-2=B \to B=-2$，\quad $s$ の係数：$1=A \to A=1$

$\mathcal{L}^{-1} = e^{-3t} - 2te^{-3t}$

(2) 平方完成：$s^2-4s+13=(s-2)^2+3^2$

分子を $s=(s-2)+2$ と分解：

$\dfrac{s}{(s-2)^2+9} = \dfrac{s-2}{(s-2)^2+9}+\dfrac{2}{(s-2)^2+9}$

$\mathcal{L}^{-1} = e^{2t}\cos 3t + \dfrac{2}{3}e^{2t}\sin 3t$
}

\mondai{92}{
次の関数の逆ラプラス変換を求めよ．
(1) $\dfrac{2s^2-5s-6}{(s+2)(s-1)(s-2)}$ \quad
(2) $\dfrac{1}{s^2(s+1)}$
\hfill {\scriptsize [教p.60(問19)]}
}

\kotae{
(1) $\dfrac{2s^2-5s-6}{(s+2)(s-1)(s-2)} = \dfrac{A}{s+2}+\dfrac{B}{s-1}+\dfrac{C}{s-2}$ とおく

通分：$2s^2-5s-6 = A(s-1)(s-2)+B(s+2)(s-2)+C(s+2)(s-1)$

$s=-2$：$8+10-6=12=A(-3)(-4) \to A=1$

$s=1$：$2-5-6=-9=B(3)(-1) \to B=3$

$s=2$：$8-10-6=-8=C(4)(1) \to C=-2$

$\mathcal{L}^{-1} = e^{-2t}+3e^t-2e^{2t}$

(2) $\dfrac{1}{s^2(s+1)} = \dfrac{A}{s}+\dfrac{B}{s^2}+\dfrac{C}{s+1}$ とおく

通分：$1=As(s+1)+B(s+1)+Cs^2$

$s=0$：$1=B \to B=1$，\quad $s=-1$：$1=C \to C=1$

$s^2$ の係数：$0=A+C \to A=-1$

$\mathcal{L}^{-1} = -1+t+e^{-t}$
}
