\mondai{161}{
次の関数のスペクトルを求めよ．
$f(x) = |x| \quad (|x| \leq 1)$，\quad $f(x+2)=f(x)$
\hfill {\scriptsize [教p.101(問10)]}
}

\kotae{
周期 2（$l=1$）なのでフーリエ係数：$c_n = \dfrac{1}{2}\displaystyle\int_{-1}^{1}|x|e^{-in\pi x}\,dx$

$c_0 = \dfrac{1}{2}\displaystyle\int_{-1}^{1}|x|\,dx = \displaystyle\int_{0}^{1}x\,dx = \dfrac{1}{2}$

$n\neq 0$：$|x|$ が偶関数なので虚部は 0，$c_n = \displaystyle\int_{0}^{1}x\cos(n\pi x)\,dx$

部分積分：$p=x$，$dq=\cos(n\pi x)dx$，$dp=dx$，$q=\dfrac{\sin(n\pi x)}{n\pi}$

$= \left[\dfrac{x\sin(n\pi x)}{n\pi}\right]_0^1 - \dfrac{1}{n\pi}\displaystyle\int_0^1\sin(n\pi x)\,dx = 0 + \dfrac{1}{n\pi}\left[\dfrac{\cos(n\pi x)}{n\pi}\right]_0^1 = \dfrac{(-1)^n-1}{n^2\pi^2}$

スペクトル（$\omega_0=\pi$）：$S_f(\omega) = \begin{cases} \dfrac{1}{2} & (\omega=0) \\[4pt] \dfrac{|(-1)^n-1|}{n^2\pi^2} & (\omega=n\pi,\ n\neq 0) \end{cases}$
}

\mondai{162}{
次の関数のスペクトルを求めよ．
$f(x) = \begin{cases} |x| & (|x| \leq 1) \\ 0 & (|x| > 1) \end{cases}$
\hfill {\scriptsize [教p.102(問11)]}
}

\kotae{
非周期関数なのでフーリエ変換を用いる：$F(\omega) = \displaystyle\int_{-1}^{1}|x|e^{-i\omega x}\,dx$

$|x|$ が偶関数なので $F(\omega) = 2\displaystyle\int_{0}^{1}x\cos(\omega x)\,dx$

部分積分：$p=x$，$dq=\cos(\omega x)dx$，$dp=dx$，$q=\dfrac{\sin(\omega x)}{\omega}$

$= 2\left[\dfrac{x\sin(\omega x)}{\omega}\right]_0^1 - \dfrac{2}{\omega}\displaystyle\int_0^1\sin(\omega x)\,dx = \dfrac{2\sin\omega}{\omega} + \dfrac{2}{\omega}\left[\dfrac{\cos(\omega x)}{\omega}\right]_0^1$

$= \dfrac{2\sin\omega}{\omega} + \dfrac{2(\cos\omega-1)}{\omega^2} = \dfrac{2(\omega\sin\omega+\cos\omega-1)}{\omega^2}$

スペクトル：$S_f(\omega) = \dfrac{2|\omega\sin\omega+\cos\omega-1|}{\omega^2} \quad (\omega\neq 0)$
}
