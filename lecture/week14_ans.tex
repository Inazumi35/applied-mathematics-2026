\begin{frame}{練習問題2　（p.\,104）}

    $(2) g(x) = { 1 - x/2  (0 \leq x < 2),  0  (x < 0, x \geq 2) }$\\
    \item[問2] 次の関数について、(1) はフーリエ正弦変換、(2) はフーリエ余弦変換をそれぞれ求めよ
    $(1) f(x) = { \sin x  (|x| \leq \pi),  0  (|x| > \pi) }$\\
    $(2) g(x) = { \cos x  (|x| \leq \pi/2),  0  (|x| > \pi/2) }$\\
  \end{description}
  \end{block}
\end{frame}

\begin{frame}{問10　解答}
  \begin{block}{}
  \begin{itemize}\setlength{\itemsep}{2pt}
    \item 周期 $T=2, l=1$ の偶関数 $: b_{n}=0$
    \item $a_{0} = 2\int_{0}^1(1-x)dx = 1$
    \item $a_{n} = 2\int_{0}^1(1-x)\cos(n\pi x)dx = 2(1-(-1)^n)/(n^{2}\pi^{2})$
    \item n が奇数 $: a_{n} = 4/(n^{2}\pi^{2}), n$ が偶数 $: a_{n} = 0$
    \item 複素フーリエ係数 $: c_{0} = a_{0}/2 = 1/2, |c_{n}| = a_{n}/2 = 2/(n^{2}\pi^{2})$ （奇数 n ）
  \end{itemize}
  \end{block}
  \begin{exampleblock}{答え}
    \[
    スペクトル: |c_{0}|=1/2,  |c_{n}|=2/(n^{2}\pi^{2})（n=1,3,5,...）,  0（n=2,4,6,...）
  \]
  \end{exampleblock}
\end{frame}

\begin{frame}{問11　解答}
  \begin{block}{}
  \begin{itemize}\setlength{\itemsep}{2pt}
    \item 非周期関数 $ \to $ フーリエ変換でスペクトルを求める
    \item $f(x)$ は偶関数 $: F(u) = 2\int_{0}^1(1-x)\cos(ux)dx$
    \item $= 2{[(1-x)\sin(ux)/u]_{0}^1 + (1/u)\int_{0}^1 \sin(ux)dx}$
    \item $= 2(1/u)[-\cos(ux)/u]_{0}^1 = 2(1-\cos u)/u^{2} = 4\sin^{2}(u/2)/u^{2}$
  \end{itemize}
  \end{block}
  \begin{exampleblock}{答え}
    \[
    スペクトル S(u) = |F(u)|/\pi = 4\sin^{2}(u/2)/(\pi u^{2})（連続スペクトル）
  \]
  \end{exampleblock}
\end{frame}
