\begin{frame}{定義：線スペクトル}
  \begin{block}{}
  周期 $2l$ の関数 $f(x) = \sum_{n=-\infty}^\infty c_{n} e^{in\pi x/l}$ に対し、\\[4pt]
  角周波数 $\omega_{n} = n\pi/l$ （とびとびの値）における $|c_{n}|$ を\\[4pt]
  \textbf{線スペクトル}（振幅スペクトル）という
  \end{block}
  {\footnotesize \textcolor{gray}{周期関数 $\to$ 離散的な $\omega_{n}$ に対応する離散スペクトル}}
\end{frame}

\begin{frame}{定義：連続スペクトル}
  \begin{block}{}
  非周期関数 $f(x)$ のフーリエ変換を $F(\omega)$ とするとき\\[4pt]
  \[
    S(\omega) = |F(\omega)| / \pi
  \]
  を \textbf{連続スペクトル}（振幅スペクトル）という
  \end{block}
  {\footnotesize \textcolor{gray}{$l\to\infty$ の極限で線スペクトルの密度 $|c_{n}|/\Delta\omega \to |F(\omega)|/(2\pi)$；両側を片側に集約して $S(\omega)=|F(\omega)|/\pi$}}
\end{frame}

\begin{frame}{例2　（p.\,100）}
  \begin{exampleblock}{問題}
    周期4の矩形波のスペクトルを求めよ\\[2pt]
    $f(x) = \begin{cases} 1 & (-1 < x < 1) \\ 1/2 & (x = 1, 3) \\ 0 & (1 < x < 3) \end{cases},  f(x+4) = f(x)$\\[2pt]
  \end{exampleblock}
\end{frame}

\begin{frame}{例2　解答}
  $l=2,\; \omega_{n}=n\pi/2$ より $c_{n} = \dfrac{1}{4}\!\int_{-1}^{1} e^{-i\omega_{n}x}\,dx$\\[4pt]
  $\omega_{n}=0\ (n=0)$：$c_{0} = \dfrac{1}{2}$\\[4pt]
  $\omega_{n}\neq 0\ (n\neq 0)$：$c_{n} = \dfrac{\sin(n\pi/2)}{n\pi}$ より\\[6pt]
  \quad $\omega_{n}=\dfrac{\pi}{2},\dfrac{3\pi}{2},\ldots$（$n$ 奇数）：$|c_{n}| = \dfrac{1}{n\pi}$\\[4pt]
  \quad $\omega_{n}=\pi, 2\pi,\ldots$（$n$ 偶数，$\neq 0$）：$c_{n} = 0$
  \\[8pt]\textcolor{red}{\textbf{答}}：線スペクトル\quad
  $|c_{0}|=\dfrac{1}{2},\quad |c_{n}|=\dfrac{1}{n\pi}$（$\omega_{n}=\dfrac{\pi}{2},\dfrac{3\pi}{2},\ldots$），$0$（その他）
\end{frame}

\begin{frame}{演習問題}
  \begin{exampleblock}{自分で解いてみよう}
  \begin{description}
    \item[問10] 次の関数のスペクトルを求めよ
    $f(x) = 1 - |x|  (|x| \leq 1),  f(x+2) = f(x)$\\
  \end{description}
  \end{exampleblock}
\end{frame}

\begin{frame}{問10　解答}
  周期 $T=2,\; l=1$，$\omega_{n}=n\pi$ の偶関数 $:\; b_{n}=0$\\[4pt]
  $c_{0} = \displaystyle\int_{0}^1(1-x)\,dx = \dfrac{1}{2}$\\[6pt]
  $a_{n} = 2\displaystyle\int_{0}^1(1-x)\cos(\omega_{n} x)\,dx = \dfrac{2(1-(-1)^n)}{n^{2}\pi^{2}}$\\[4pt]
  \quad $\omega_{n}=\pi, 3\pi,\ldots$（$n$ 奇数）$:\; a_{n} = \dfrac{4}{n^{2}\pi^{2}}$\\[4pt]
  \quad $\omega_{n}=2\pi, 4\pi,\ldots$（$n$ 偶数）$:\; a_{n} = 0$\\[4pt]
  線スペクトル $:\; |c_{n}| = a_{n} = \dfrac{4}{n^{2}\pi^{2}} = \dfrac{4}{\omega_{n}^{2}}$（$n$ 奇数）
  \\[6pt]\textcolor{red}{\textbf{答}}：スペクトル\quad $|c_{0}|=\dfrac{1}{2}$，$|c_{n}|=\dfrac{4}{\omega_{n}^{2}}$（$\omega_{n}=\pi, 3\pi,\ldots$），$0$（その他）
\end{frame}

\begin{frame}{例3　（p.\,101）}
  \begin{exampleblock}{問題}
    次の関数のスペクトルを求めよ\\[2pt]
    $f(x) = \begin{cases} 1 & (|x| < 1) \\ 1/2 & (|x| = 1) \\ 0 & (|x| > 1) \end{cases}$\\[2pt]
  \end{exampleblock}
\end{frame}

\begin{frame}{例3　解答}
  非周期関数 $\to$ フーリエ変換 $F(\omega) = \int_{-\infty}^{\infty} f(x) e^{-i\omega x} dx$\\[4pt]
  $F(\omega) = \int_{-1}^{1} e^{-i\omega x} dx = [-e^{-i\omega x}/(i\omega)]_{-1}^{1}$\\[4pt]
  $\hphantom{F(\omega)} = (e^{i\omega} - e^{-i\omega})/(i\omega) = 2\sin\omega/\omega$\\[4pt]
  $\omega=0$：$F(0)=\displaystyle\int_{-1}^{1}dx=2$ より $S(\omega)=\dfrac{2}{\pi}$\\[4pt]
  $\omega> 0$：$S(\omega) = \dfrac{|F(\omega)|}{\pi} = \dfrac{2\sin\omega}{\pi\omega}$
  \\[6pt]\textcolor{red}{\textbf{答}}：$\omega=0$のとき$S(\omega)=\dfrac{2}{\pi}$，$\omega>0$のとき$S(\omega)=\dfrac{2\sin\omega}{\pi\omega}$
\end{frame}

\begin{frame}{演習問題}
  \begin{exampleblock}{自分で解いてみよう}
  \begin{description}
    \item[問11] 次の関数のスペクトルを求めよ
    $f(x) = \begin{cases} 1-|x| & (|x| \leq 1) \\ 0 & (|x| > 1) \end{cases}$\\
  \end{description}
  \end{exampleblock}
\end{frame}

\begin{frame}{問11　解答}
  非周期関数 $\to$ フーリエ変換でスペクトルを求める\\[4pt]
  $f(x)$ は偶関数 $:\; F(\omega) = 2\displaystyle\int_{0}^1(1-x)\cos(\omega x)\,dx = \dfrac{2(1-\cos\omega)}{\omega^{2}}$\\[6pt]
  $\omega=0$：$F(0)=2\displaystyle\int_0^1(1-x)\,dx=1$ より $S(\omega)=\dfrac{1}{\pi}$\\[4pt]
  $\omega>0$：$S(\omega)=\dfrac{F(\omega)}{\pi}=\dfrac{2(1-\cos\omega)}{\pi\omega^{2}}$
  \\[6pt]\textcolor{red}{\textbf{答}}：$\omega=0$のとき$S(\omega)=\dfrac{1}{\pi}$，$\omega>0$のとき$S(\omega)=\dfrac{2(1-\cos\omega)}{\pi\omega^{2}}$
\end{frame}
