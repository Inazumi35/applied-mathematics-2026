\begin{frame}{第11回　複素フーリエ級数}
  \begin{block}{今回の内容}
  複素フーリエ級数
  \end{block}
  \vfill\hfill{\scriptsize \textcolor{gray}{第3章 フーリエ解析　§1}}
\end{frame}

\begin{frame}{定義：複素フーリエ級数（周期 $2l$）}
  \begin{block}{}
  \[
    f(x) = \sum_{n=-\infty}^{\infty} c_{n} e^{in\pi x/l}, \quad
    c_{n} = \frac{1}{2l} \int_{-l}^{l} f(x) e^{-in\pi x/l} dx
  \]
  \end{block}
  {\footnotesize\textcolor{gray}{$e^{i\theta}=\cos\theta+i\sin\theta$（オイラーの公式）より，$e^{in\pi x/l}$ は $\cos/\sin$ の代わりとして機能する}}
\end{frame}

\begin{frame}{例題6　（p.\,88）}
  \begin{exampleblock}{問題}
    次の周期 4 の関数 $f(x)$ の複素フーリエ級数を求めよ\\[2pt]
    $f(x) = x + 1  (-2 \leq x < 2),  f(x+4) = f(x)$\\[2pt]
  \end{exampleblock}
\end{frame}

\begin{frame}{例題6　解答（1/3）}
  $l=2$ より $c_{n} = \dfrac{1}{4}\displaystyle\int_{-2}^{2}(x+1)e^{-in\pi x/2}dx$\\[6pt]
  \textbf{$n=0$ のとき：}\\[4pt]
  $c_{0} = \dfrac{1}{4}\!\left[\dfrac{x^{2}}{2}+x\right]_{-2}^{2}
    = \dfrac{1}{4}\!\left\{(2+2)-(2-2)\right\} = \dfrac{1}{4}\cdot 4 = 1$\\[6pt]
  \textbf{$n\neq 0$ のとき：}部分積分 $u=x+1,\;dv=e^{-in\pi x/2}dx$ より\\[2pt]
  \hspace{2em}$du=dx,\quad v = \dfrac{-2}{in\pi}e^{-in\pi x/2}$\\[4pt]
  $\displaystyle\int_{-2}^{2}(x+1)e^{-in\pi x/2}dx
    = \!\left[\dfrac{-2(x+1)}{in\pi}e^{-in\pi x/2}\right]_{-2}^{2}
    + \dfrac{2}{in\pi}\int_{-2}^{2}e^{-in\pi x/2}dx$
\end{frame}

\begin{frame}{例題6　解答（2/3）}
  \textbf{第2項の計算：}\\[4pt]
  $\displaystyle\int_{-2}^{2}e^{-in\pi x/2}dx
    = \!\left[\dfrac{-2}{in\pi}e^{-in\pi x/2}\right]_{-2}^{2}
    = \dfrac{-2}{in\pi}\!\left(e^{-in\pi}-e^{in\pi}\right)$\\[6pt]
  オイラーの公式 $e^{i\theta}-e^{-i\theta}=2i\sin\theta$ より\\[2pt]
  \hspace{2em}$e^{-in\pi}-e^{in\pi} = -(e^{in\pi}-e^{-in\pi}) = -2i\sin(n\pi)$\\[6pt]
  $n$ は整数なので $\sin(n\pi)=0$，よって\\[2pt]
  \hspace{2em}$\dfrac{-2}{in\pi}\cdot(-2i\sin n\pi) = \dfrac{-2}{in\pi}\cdot 0 = 0$\\[4pt]
  $\therefore$ 第2項 $= \dfrac{2}{in\pi}\cdot 0 = 0$
\end{frame}

\begin{frame}{例題6　解答（3/3）}
  \textbf{第1項}（$e^{\pm in\pi}=(-1)^{n}$ を代入）：\\[4pt]
  $x=2$ のとき：$\dfrac{-2(2+1)}{in\pi}e^{-in\pi} = \dfrac{-6}{in\pi}\cdot(-1)^{n} = \dfrac{-6(-1)^{n}}{in\pi}$\\[4pt]
  $x=-2$ のとき：$\dfrac{-2(-2+1)}{in\pi}e^{in\pi} = \dfrac{2}{in\pi}\cdot(-1)^{n} = \dfrac{2(-1)^{n}}{in\pi}$\\[4pt]
  $\therefore\;\!\left[\dfrac{-2(x+1)}{in\pi}e^{-in\pi x/2}\right]_{-2}^{2}
    = \dfrac{-6(-1)^{n}}{in\pi} - \dfrac{2(-1)^{n}}{in\pi}
    = \dfrac{-8(-1)^{n}}{in\pi}$\\[6pt]
  $c_{n} = \dfrac{1}{4}\cdot\dfrac{-8(-1)^{n}}{in\pi} = \dfrac{-2(-1)^{n}}{in\pi} = \dfrac{2i(-1)^{n}}{n\pi}$
  \hspace{4pt}{\footnotesize\textcolor{gray}{$-1/i=i$ を利用}}
  \\[8pt]\textcolor{red}{\textbf{答}}：$f(x) = 1 + \displaystyle\sum_{\substack{n=-\infty \\ n\neq 0}}^{\infty} \dfrac{2i(-1)^{n}}{n\pi}\,e^{in\pi x/2}$
\end{frame}

\begin{frame}{演習問題}
  \begin{exampleblock}{自分で解いてみよう}
  \begin{description}
    \item[問6] 次の周期 2 の関数 $f(x), g(x)$ の複素フーリエ級数を求めよ
    $(1) f(x) = \begin{cases} 0 & (-1 \leq x < 0) \\ 1 & (0 \leq x < 1) \end{cases},  f(x+2) = f(x)$\\
    $(2) g(x) = \begin{cases} 0 & (-1 \leq x < 0) \\ x & (0 \leq x < 1) \end{cases},  g(x+2) = g(x)$\\
    \item[問7] 次の周期 4 の関数 $f(x), g(x)$ の複素フーリエ級数を求めよ
    $(1) f(x) = \begin{cases} 2 & (-2 \leq x < 0) \\ 0 & (0 \leq x < 2) \end{cases},  f(x+4) = f(x)$\\
    $(2) g(x) = \begin{cases} 0 & (-2 \leq x < 0) \\ 2-x & (0 \leq x < 2) \end{cases},  g(x+4) = g(x)$\\
  \end{description}
  \end{exampleblock}
\end{frame}

\begin{frame}{問6(1)　解答（1/2）}
  $f(x)=0\;(-1\leq x<0)$ より，積分は $[0,1]$ のみ寄与\\[4pt]
  $l=1$：$c_{n} = \dfrac{1}{2}\displaystyle\int_{-1}^{1}f(x)e^{-in\pi x}dx
    = \dfrac{1}{2}\displaystyle\int_{0}^{1}e^{-in\pi x}\,dx$\\[6pt]
  \textbf{$n=0$：}$c_{0} = \dfrac{1}{2}\displaystyle\int_{0}^{1}dx = \dfrac{1}{2}$\\[6pt]
  \textbf{$n\neq 0$：}
  $c_{n} = \dfrac{1}{2}\!\left[\dfrac{-e^{-in\pi x}}{in\pi}\right]_{0}^{1}
    = \dfrac{-1}{2in\pi}\!\left(e^{-in\pi}-e^{0}\right)
    = \dfrac{1-e^{-in\pi}}{2in\pi}$
\end{frame}

\begin{frame}{問6(1)　解答（2/2）}
  $e^{-in\pi}$ の計算（オイラーの公式 $e^{i\theta}=\cos\theta+i\sin\theta$ より）：\\[4pt]
  $e^{-in\pi} = \cos(-n\pi)+i\sin(-n\pi) = \cos(n\pi)-i\sin(n\pi)$\\[2pt]
  $n$ は整数なので $\sin(n\pi)=0$，よって $e^{-in\pi} = \cos(n\pi) = (-1)^{n}$\\[6pt]
  代入：$c_{n} = \dfrac{1-(-1)^{n}}{2in\pi}$\\[6pt]
  $\dfrac{1}{in}=\dfrac{-i}{n}$ を利用：\\[2pt]
  $c_{n} = \dfrac{-i\bigl(1-(-1)^{n}\bigr)}{2n\pi} = \dfrac{i\bigl((-1)^{n}-1\bigr)}{2n\pi}$
  \\[8pt]\textcolor{red}{\textbf{答}}：
  $(1)\; f(x) = \dfrac{1}{2} + \displaystyle\sum_{\substack{n=-\infty \\ n\neq 0}}^{\infty} \dfrac{i((-1)^{n}-1)}{2n\pi}\,e^{in\pi x}$
\end{frame}

\begin{frame}{問6(2)　解答（1/2）}
  $g(x)=0\;(-1\leq x<0)$ より，積分は $[0,1]$ のみ寄与\\[4pt]
  $l=1$：$c_{n} = \dfrac{1}{2}\displaystyle\int_{0}^{1}x\,e^{-in\pi x}\,dx$\\[4pt]
  \textbf{$n=0$：}$c_{0} = \dfrac{1}{2}\displaystyle\int_{0}^{1}x\,dx = \dfrac{1}{4}$\\[4pt]
  \textbf{$n\neq 0$：}部分積分 $u=x,\;dv=e^{-in\pi x}dx$\\[2pt]
  \hspace{2em}$du=dx,\quad v=\dfrac{-e^{-in\pi x}}{in\pi}$\\[4pt]
  $= \dfrac{1}{2}\!\left\{\!\left[\dfrac{-xe^{-in\pi x}}{in\pi}\right]_{0}^{1}
    + \dfrac{1}{in\pi}\displaystyle\int_{0}^{1}e^{-in\pi x}\,dx\right\}$\\[4pt]
  $= \dfrac{1}{2}\!\left\{
      \underbrace{\dfrac{-e^{-in\pi}}{in\pi}}_{A}
      + \dfrac{1}{in\pi}\underbrace{\left[\dfrac{-e^{-in\pi x}}{in\pi}\right]_{0}^{1}}_{B}
    \right\}$
\end{frame}

\begin{frame}{問6(2)　解答（2/2）}
  $A = \dfrac{-e^{-in\pi}}{in\pi} = \dfrac{-(-1)^{n}}{in\pi} = \dfrac{i(-1)^{n}}{n\pi}$
  \hspace{8pt}{\footnotesize\textcolor{gray}{$-1/(in) = i/n$ を利用}}\\[6pt]
  $B = \left[\dfrac{-e^{-in\pi x}}{in\pi}\right]_{0}^{1}
    = \dfrac{-1}{in\pi}\!\left(e^{-in\pi}-e^{0}\right) = \dfrac{1-(-1)^{n}}{in\pi}$
  なので\\[4pt]
  $(in\pi)^{2} = -n^{2}\pi^{2}$ より
  $\dfrac{1}{in\pi}\cdot B = \dfrac{1-(-1)^{n}}{(in\pi)^{2}} = \dfrac{1-(-1)^{n}}{-n^{2}\pi^{2}} = \dfrac{(-1)^{n}-1}{n^{2}\pi^{2}}$\\[4pt]
  $c_{n} = \dfrac{1}{2}\!\left(\dfrac{i(-1)^{n}}{n\pi} + \dfrac{(-1)^{n}-1}{n^{2}\pi^{2}}\right)
    = \dfrac{i(-1)^{n}}{2n\pi} + \dfrac{(-1)^{n}-1}{2n^{2}\pi^{2}}$
  \\[8pt]\textcolor{red}{\textbf{答}}：
  $(2)\; g(x) = \dfrac{1}{4} + \displaystyle\sum_{\substack{n=-\infty \\ n\neq 0}}^{\infty} \!\left[\dfrac{i(-1)^{n}}{2n\pi}+\dfrac{(-1)^{n}-1}{2n^{2}\pi^{2}}\right] e^{in\pi x}$\\
\end{frame}

\begin{frame}{問7(1)　解答}
  $f(x)=0\;(0\leq x<2)$ より，積分は $[-2,0]$ のみ寄与\\[2pt]
  $l=2$：$c_{n} = \dfrac{1}{4}\displaystyle\int_{-2}^{0}2\,e^{-in\pi x/2}\,dx
    = \dfrac{1}{2}\displaystyle\int_{-2}^{0}e^{-in\pi x/2}\,dx$\\[4pt]
  \textbf{$n=0$：}$c_{0} = \dfrac{1}{2}\displaystyle\int_{-2}^{0}dx = 1$\\[2pt]
  \textbf{$n\neq 0$：}
  $c_{n} = \dfrac{1}{2}\!\left[\dfrac{-2\,e^{-in\pi x/2}}{in\pi}\right]_{-2}^{0}$\\[2pt]
  \hspace{1em}$x=0$のとき $\dfrac{-2}{in\pi}e^{0}=\dfrac{-2}{in\pi}$，
  $x=-2$のとき $\dfrac{-2}{in\pi}e^{in\pi}=\dfrac{-2(-1)^{n}}{in\pi}$\\[2pt]
  $c_{n} = \dfrac{1}{2}\!\left(\dfrac{-2}{in\pi}-\dfrac{-2(-1)^{n}}{in\pi}\right)
    = \dfrac{-1}{in\pi}(1-(-1)^{n}) = \dfrac{i(1-(-1)^{n})}{n\pi}$
  \hspace{4pt}{\footnotesize\textcolor{gray}{$1/(in)=-i/n$ を利用}}
  \\[4pt]\textcolor{red}{\textbf{答}}：
  $(1)\; f(x) = 1 + \displaystyle\sum_{\substack{n=-\infty \\ n\neq 0}}^{\infty} \dfrac{i(1-(-1)^{n})}{n\pi}\,e^{in\pi x/2}$\\
\end{frame}

\begin{frame}{問7(2)　解答（1/2）}
  $g(x)=0\;(-2\leq x<0)$ より，積分は $[0,2]$ のみ寄与\\[4pt]
  $l=2$：$c_{n} = \dfrac{1}{4}\displaystyle\int_{0}^{2}(2-x)\,e^{-in\pi x/2}\,dx$\\[4pt]
  \textbf{$n=0$：}$c_{0} = \dfrac{1}{4}\!\left[2x-\dfrac{x^{2}}{2}\right]_{0}^{2}
    = \dfrac{1}{4}\!\left\{(4-2)-0\right\} = \dfrac{1}{2}$\\[4pt]
  \textbf{$n\neq 0$：}部分積分 $u=2-x,\;dv=e^{-in\pi x/2}dx$\\[2pt]
  \hspace{2em}$du=-dx,\quad v=\dfrac{-2}{in\pi}e^{-in\pi x/2}$\\[4pt]
  $\displaystyle\int_{0}^{2}(2-x)e^{-in\pi x/2}dx
    = \underbrace{\!\left[\dfrac{-2(2-x)}{in\pi}e^{-in\pi x/2}\right]_{0}^{2}}_{A}
      - \underbrace{\int_{0}^{2}\dfrac{2}{in\pi}e^{-in\pi x/2}dx}_{B}$
\end{frame}

\begin{frame}{問7(2)　解答（2/3）}
  \textbf{A の計算：}\\[2pt]
  $x=2$のとき $2-x=0$ より $\dfrac{-2\cdot 0}{in\pi}e^{-in\pi}=0$\\[2pt]
  $x=0$のとき $\dfrac{-2\cdot 2}{in\pi}e^{0}=\dfrac{-4}{in\pi}$\\[2pt]
  $A = 0 - \dfrac{-4}{in\pi} = \dfrac{4}{in\pi}$\\[6pt]
  \textbf{B の計算：}$\left[\dfrac{-2\,e^{-in\pi x/2}}{in\pi}\right]_{0}^{2}$ を評価：\\[2pt]
  $x=2$のとき $\dfrac{-2\,e^{-in\pi}}{in\pi}=\dfrac{-2(-1)^{n}}{in\pi}$，
  $x=0$のとき $\dfrac{-2\,e^{0}}{in\pi}=\dfrac{-2}{in\pi}$\\[2pt]
  差 $= \dfrac{-2(-1)^{n}}{in\pi}-\dfrac{-2}{in\pi} = \dfrac{-2\bigl((-1)^{n}-1\bigr)}{in\pi}$
  なので\\[4pt]
  $B = \dfrac{2}{in\pi}\cdot\dfrac{-2((-1)^{n}-1)}{in\pi}
    = \dfrac{-4((-1)^{n}-1)}{(in\pi)^{2}} = \dfrac{4((-1)^{n}-1)}{n^{2}\pi^{2}}$
  \hspace{4pt}{\footnotesize\textcolor{gray}{$(in\pi)^{2}=-n^{2}\pi^{2}$}}
\end{frame}

\begin{frame}{問7(2)　解答（3/3）}
  A，B を代入：\\[4pt]
  $\displaystyle\int_{0}^{2}(2-x)e^{-in\pi x/2}dx = A - B
    = \dfrac{4}{in\pi} - \dfrac{4((-1)^{n}-1)}{n^{2}\pi^{2}}
    = \dfrac{4}{in\pi} + \dfrac{4(1-(-1)^{n})}{n^{2}\pi^{2}}$\\[6pt]
  $c_{n} = \dfrac{1}{4}\!\left(\dfrac{4}{in\pi} + \dfrac{4(1-(-1)^{n})}{n^{2}\pi^{2}}\right)
    = \dfrac{1}{in\pi} + \dfrac{1-(-1)^{n}}{n^{2}\pi^{2}}
    = -\dfrac{i}{n\pi} + \dfrac{1-(-1)^{n}}{n^{2}\pi^{2}}$
  \hspace{4pt}{\footnotesize\textcolor{gray}{$1/(in)=-i/n$ を利用}}
  \\[8pt]\textcolor{red}{\textbf{答}}：
  $(2)\; g(x) = \dfrac{1}{2} + \displaystyle\sum_{\substack{n=-\infty \\ n\neq 0}}^{\infty} \!\left[-\dfrac{i}{n\pi}+\dfrac{1-(-1)^{n}}{n^{2}\pi^{2}}\right] e^{in\pi x/2}$\\
\end{frame}
