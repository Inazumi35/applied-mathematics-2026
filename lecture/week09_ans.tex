  一般の周期関数のフーリエ級数（１）
  \end{block}
  \vfill\hfill{\scriptsize \textcolor{gray}{第3章 フーリエ解析　§1}}
\end{frame}

\begin{frame}{定義：フーリエ級数（周期 T = 2l）}
  \begin{block}{}
  \[
    f(x) = c_{0} + \sum_{n=1}^\infty \left(a_{n}\cos\frac{n\pi x}{l} + b_{n}\sin\frac{n\pi x}{l}\right)
  \]
  \medskip\textbf{フーリエ係数}：
  \begin{align*}
    c_{0} &= \frac{1}{2l} \int_{-l}^{l} f(x)\,dx \\[4pt]
    a_{n} &= \frac{1}{l} \int_{-l}^{l} f(x)\cos\frac{n\pi x}{l}\,dx \quad(n=1,2,\ldots) \\[4pt]
    b_{n} &= \frac{1}{l} \int_{-l}^{l} f(x)\sin\frac{n\pi x}{l}\,dx \quad(n=1,2,\ldots)
  \end{align*}
  \end{block}
\end{frame}

\begin{frame}{フーリエ余弦級数・フーリエ正弦級数}
  \begin{block}{偶関数 $f(-x)=f(x)$ のとき（フーリエ余弦級数）}
    $b_{n}=0$ なので
    \[
      f(x) = c_{0} + \sum_{n=1}^{\infty} a_{n}\cos\frac{n\pi x}{l},\quad
      a_{n} = \frac{2}{l}\int_{0}^{l} f(x)\cos\frac{n\pi x}{l}\,dx
    \]
  \end{block}
  \begin{block}{奇関数 $f(-x)=-f(x)$ のとき（フーリエ正弦級数）}
    $c_{0}=0,\;a_{n}=0$ なので
    \[
      f(x) = \sum_{n=1}^{\infty} b_{n}\sin\frac{n\pi x}{l},\quad
      b_{n} = \frac{2}{l}\int_{0}^{l} f(x)\sin\frac{n\pi x}{l}\,dx
    \]
  \end{block}
\end{frame}

\begin{frame}{例題2　（p.\,81）}
  \begin{exampleblock}{問題}
    $f(x) = \begin{cases} 0 & (-1 \leq x < 0) \\ 1 & (0 \leq x < 1) \end{cases},\; f(x+2) = f(x)$ のフーリエ級数を求めよ\\[2pt]
  \end{exampleblock}
\end{frame}

\begin{frame}{例題2　解答（1/2）}
  $l=1$．$f(x)$ は偶関数でも奇関数でもない．\\[4pt]
  $c_{0} = \dfrac{1}{2}\displaystyle\int_{-1}^{1} f(x)\,dx = \dfrac{1}{2}\int_{0}^{1} 1\,dx = \dfrac{1}{2}$\\[6pt]
  $a_{n} = \displaystyle\int_{-1}^{1} f(x)\cos(n\pi x)\,dx = \int_{0}^{1}\cos(n\pi x)\,dx = \left[\dfrac{\sin(n\pi x)}{n\pi}\right]_{0}^{1} = 0$\\[6pt]
  $b_{n} = \displaystyle\int_{-1}^{1} f(x)\sin(n\pi x)\,dx = \int_{0}^{1}\sin(n\pi x)\,dx = \left[-\dfrac{\cos(n\pi x)}{n\pi}\right]_{0}^{1} = \dfrac{1-(-1)^{n}}{n\pi}$
\end{frame}

\begin{frame}{例題2　解答（2/2）}
  $n$ が奇数のとき $b_{n} = \dfrac{2}{n\pi}$，偶数のとき $b_{n} = 0$
  \\[8pt]\textcolor{red}{\textbf{答}}：$f(x) = \dfrac{1}{2} + \dfrac{2}{\pi}\displaystyle\sum_{k=0}^{\infty} \dfrac{\sin(2k+1)\pi x}{2k+1} = \dfrac{1}{2} + \dfrac{2}{\pi}\!\left(\sin\pi x + \dfrac{\sin 3\pi x}{3} + \dfrac{\sin 5\pi x}{5} + \cdots\right)$
\end{frame}

\begin{frame}{演習問題}
  \begin{exampleblock}{自分で解いてみよう}
  \begin{description}
    \item[問3] 次の関数のフーリエ級数を求めよ
    $(1) f(x) = \begin{cases} 1 & (-1 \leq x < 0) \\ 0 & (0 \leq x < 1) \end{cases},  f(x+2) = f(x)$\\
    $(2) g(x) = \begin{cases} 2 & (-2 \leq x < 0) \\ 4 & (0 \leq x < 2) \end{cases},  g(x+4) = g(x)$\\
    $(3) h(x) = \begin{cases} 2x+1 & (-1/2 \leq x < 0) \\ 0 & (0 \leq x < 1/2) \end{cases},  h(x+1) = h(x)$\\
  \end{description}
  \end{exampleblock}
\end{frame}

\begin{frame}{問3(1)　解答}
  $(1)\; l=1:\; c_{0}=\dfrac{1}{2},\; a_{n}=\displaystyle\int_{-1}^{0}\cos(n\pi x)\,dx=0$\\[4pt]
  $b_{n}=\displaystyle\int_{-1}^{0}\sin(n\pi x)\,dx=\dfrac{(-1)^n-1}{n\pi}$
  \\[8pt]\textcolor{red}{\textbf{答}}：$f(x) = \dfrac{1}{2} - \dfrac{2}{\pi}\displaystyle\sum_{k=0}^{\infty} \dfrac{\sin(2k+1)\pi x}{2k+1} = \dfrac{1}{2} - \dfrac{2}{\pi}\!\left(\sin \pi x + \dfrac{\sin 3\pi x}{3} + \cdots\right)$\\
\end{frame}

\begin{frame}{問3(2)　解答}
  $(2)\; l=2:\; c_{0}=3,\; a_{n}=0$\\[4pt]
  $b_{n}=\dfrac{1}{2}\!\left[2\displaystyle\int_{-2}^{0}\sin\dfrac{n\pi x}{2}\,dx + 4\int_{0}^{2}\sin\dfrac{n\pi x}{2}\,dx\right] = \dfrac{2(1-(-1)^n)}{n\pi}$
  \\[8pt]\textcolor{red}{\textbf{答}}：$g(x) = 3 + \dfrac{4}{\pi}\displaystyle\sum_{k=0}^{\infty} \dfrac{\sin\frac{(2k+1)\pi x}{2}}{2k+1} = 3 + \dfrac{4}{\pi}\!\left(\sin\dfrac{\pi x}{2} + \dfrac{\sin\frac{3\pi x}{2}}{3} + \cdots\right)$\\
\end{frame}

\begin{frame}{問3(3)　解答}
  $(3)\; l=\dfrac{1}{2}:\; c_{0}=\dfrac{1}{4},\; a_{n}=\dfrac{1-(-1)^n}{n^{2}\pi^{2}},\; b_{n}=-\dfrac{1}{n\pi}$
  \\[8pt]\textcolor{red}{\textbf{答}}：
  $(3)\; h(x) = \dfrac{1}{4} + \displaystyle\sum_{n=1}^\infty\!\left[\dfrac{1-(-1)^n}{n^{2}\pi^{2}}\cos(2n\pi x) - \dfrac{1}{n\pi}\sin(2n\pi x)\right]$\\
\end{frame}

\begin{frame}{例題3　（p.\,83）}
  \begin{exampleblock}{問題}
    $f(x) = |x| \; (-1 \leq x < 1),\; f(x+2) = f(x)$ のフーリエ級数を求めよ\\[2pt]
  \end{exampleblock}
\end{frame}

\begin{frame}{例題3　解答}
  $f(x)$ は偶関数なので $b_{n} = 0$．フーリエ余弦級数を求めればよい．\\[4pt]
  $c_{0} = \dfrac{1}{2}\displaystyle\int_{-1}^{1} |x|\,dx = \dfrac{1}{2}$\\[6pt]
  $a_{n} = 2\displaystyle\int_{0}^{1} x \cos(n\pi x)\,dx = -\dfrac{2(1-(-1)^n)}{n^{2}\pi^{2}}$
  \\[8pt]\textcolor{red}{\textbf{答}}：$f(x) = \dfrac{1}{2} - \dfrac{4}{\pi^{2}}\displaystyle\sum_{k=0}^{\infty} \dfrac{\cos(2k+1)\pi x}{(2k+1)^{2}} = \dfrac{1}{2} - \dfrac{4}{\pi^{2}}\!\left(\cos \pi x + \dfrac{\cos 3\pi x}{9} + \dfrac{\cos 5\pi x}{25} + \cdots\right)$
  {\footnotesize \textcolor{gray}{\textit{$N=2, N=5$ での収束グラフあり（ p.83 ）}}}
\end{frame}
