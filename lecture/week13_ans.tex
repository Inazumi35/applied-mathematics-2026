\begin{frame}{フーリエ変換の性質}
  \begin{block}{}
  \begin{description}
    \item[I. 線形性]\mbox{}\\$F[c_{1}f_{1}(x) + c_{2}f_{2}(x)] = c_{1}F_{1}(u) + c_{2}F_{2}(u) \ (c_{1}, c_{2} は定数)$
    \item[II. 移動則（時間シフト）]\mbox{}\\$F[f(x-a)] = e^{-iau} F(u) \ (a は実数)$
    \item[III. 移動則（周波数シフト）]\mbox{}\\$F[e^{iax}f(x)] = F(u-a) \ (a は実数)$
    \item[IV. 相似性（スケーリング）]\mbox{}\\$F[f(ax)] = (1/|a|) F(u/a) \ (a は 0 でない実数)$
    \item[V. 微分則]\mbox{}\\$F[f^{(n)}(x)] = (iu)^n F(u) \ (n は自然数)$
    \item[VI. 双対微分則]\mbox{}\\$F[(-ix)^n f(x)] = F^{(n)}(u) \ (n は自然数)$
  \end{description}
  \end{block}
\end{frame}

\begin{frame}{パーセバルの等式}
  \begin{block}{}
  \[
    \int_{-\infty}^\infty |f(t)|^{2} dt = (1/2\pi) \int_{-\infty}^\infty |F(\omega)|^{2} d\omega
  \]
  \end{block}
\end{frame}

\begin{frame}{例題4　（p.\,99）}
  \begin{exampleblock}{問題}
    次の等式を証明せよ\\[2pt]
    $(1) F[e^{-a^{2}x^{2}} * e^{-b^{2}x^{2}}] = (\pi/(ab)) e^{-u^{2}(a^{2}+b^{2})/(4a^{2}b^{2})}$ 等（たたみこみのフーリエ変換と 98 ページの公式 (1) を利用）\\[2pt]
    (2) 上記の結果と比較して、畳み込み関数の閉じた形を求める\\[2pt]
  \end{exampleblock}
\end{frame}

\begin{frame}{例題4　解答}
  $(1)$ p.\,98 公式 $: F[e^{-cx^{2}}] = \sqrt{\pi/c}\cdot e^{-u^{2}/(4c)}$ より\\[4pt]
  $\hphantom{(1)\ }F[e^{-a^{2}x^{2}}] = (\sqrt{\pi}/a)e^{-u^{2}/(4a^{2})},\ F[e^{-b^{2}x^{2}}] = (\sqrt{\pi}/b)e^{-u^{2}/(4b^{2})}$\\[4pt]
  たたみこみ定理 $: F[f*g] = F(u)G(u)$ を適用\\[4pt]
  $F[e^{-a^{2}x^{2}} * e^{-b^{2}x^{2}}] = (\pi/(ab))\cdot e^{-u^{2}(1/(4a^{2})+1/(4b^{2}))} = (\pi/(ab))e^{-u^{2}(a^{2}+b^{2})/(4a^{2}b^{2})}$\\[4pt]
  $(2)$ 結果を $\sqrt{\pi/c}\cdot e^{-u^{2}/(4c)}$ の形と比較 $: c = a^{2}b^{2}/(a^{2}+b^{2})$\\[4pt]
  係数を合わせて $e^{-a^{2}x^{2}}*e^{-b^{2}x^{2}} = (\sqrt{\pi}/\sqrt{a^{2}+b^{2}})\cdot e^{-(a^{2}b^{2}/(a^{2}+b^{2}))x^{2}}$
  \\[8pt]\textcolor{red}{\textbf{答}}：証明終
\end{frame}

\begin{frame}{演習問題}
  \begin{exampleblock}{自分で解いてみよう}
  \begin{description}
    \item[問5] 性質 III および VI を証明せよ
    \item[問6] $F[f(x)] = F(u), F[g(x)] = G(u)$ のとき、次の等式を証明せよ
    $F[f(x)g(x)] = (1/(2\pi)) F(u) * G(u)$\\
    \item[問7] 96ページの性質を用いて、次の関数のフーリエ変換を求めよ
    $(1) xe^{-x^{2}/2}$\\
    $(2) x^{2}e^{-x^{2}/2}$\\
    \item[問8] 次の等式を証明せよ
    $F^{-1}[e^{-bu^{2}}] = 1/(2\sqrt{\pi b}) \cdot e^{-x^{2}/(4b)}$（b は正の定数）\\
    \item[問9] 次の等式を証明せよ
    $(1) F[e^{-x^{2}/2} * xe^{-x^{2}/2}] = -2\pi iu\cdot e^{-u^{2}}$\\
    $(2) e^{-x^{2}/2} * xe^{-x^{2}/2} = (\sqrt{\pi}/2)\cdot xe^{-x^{2}/4}$\\
  \end{description}
  \end{exampleblock}
\end{frame}

\begin{frame}{問5　解答}
  $III: F[e^{iax}f(x)] = \int e^{iax}f(x)e^{-iux}dx = \int f(x)e^{-i(u-a)x}dx = F(u-a)$\\[4pt]
  $VI: F^{(n)}(u) = d^n/du^n \int f(x)e^{-iux}dx = \int f(x)(-ix)^n e^{-iux}dx = F[(-ix)^n f(x)]$
  \\[8pt]\textcolor{red}{\textbf{答}}：証明終
\end{frame}

\begin{frame}{問6　解答}
  $g(x) = (1/2\pi)\int G(v)e^{ivx}dv$ を $F[f(x)g(x)]$ に代入\\[4pt]
  $= (1/2\pi)\int G(v)[\int f(x)e^{-i(u-v)x}dx]dv = (1/2\pi)\int G(v)F(u-v)dv = (1/2\pi)F*G$
  \\[8pt]\textcolor{red}{\textbf{答}}：証明終
\end{frame}

\begin{frame}{問7　解答}
  性質 $VI: F[(-ix)^n f(x)] = F^{(n)}(u)$\\[4pt]
  $(1) n=1: F[(-ix)e^{-x^{2}/2}] = d/du[\sqrt{2\pi}e^{-u^{2}/2}] = -\sqrt{2\pi}\cdot u\cdot e^{-u^{2}/2}$\\[4pt]
  $    F[xe^{-x^{2}/2}] = (1/(-i))\cdot(-\sqrt{2\pi}u e^{-u^{2}/2}) = -i\sqrt{2\pi} u e^{-u^{2}/2}$\\[4pt]
  $(2) n=2: (-ix)^{2} = -x^{2}, F[-x^{2}e^{-x^{2}/2}] = d^{2}/du^{2}[\sqrt{2\pi}e^{-u^{2}/2}] = \sqrt{2\pi}(u^{2}-1)e^{-u^{2}/2}$\\[4pt]
  $    F[x^{2}e^{-x^{2}/2}] = \sqrt{2\pi}(1-u^{2})e^{-u^{2}/2}$
  \\[8pt]\textcolor{red}{\textbf{答}}：
  $(1) F[xe^{-x^{2}/2}] = -i\sqrt{2\pi} u e^{-u^{2}/2}$\\
  $(2) F[x^{2}e^{-x^{2}/2}] = \sqrt{2\pi}(1-u^{2}) e^{-u^{2}/2}$\\
\end{frame}

\begin{frame}{問8　解答}
  p.98 公式 $F[e^{-ax^{2}}] = \sqrt{\pi/a}\cdot e^{-u^{2}/(4a)}$ で $b = 1/(4a)$ とおくと $a = 1/(4b)$\\[4pt]
  $F[e^{-x^{2}/(4b)}] = \sqrt{4\pi b}\cdot e^{-bu^{2}} = 2\sqrt{\pi b}\cdot e^{-bu^{2}}$\\[4pt]
  両辺を $2\sqrt{\pi b}$ で割って逆変換 $: F^{-1}[e^{-bu^{2}}] = e^{-x^{2}/(4b)}/(2\sqrt{\pi b})$
  \\[8pt]\textcolor{red}{\textbf{答}}：証明終
\end{frame}

\begin{frame}{問9　解答}
  (1) たたみこみ定理 $: F[f*g] = F(u)\cdot G(u)$\\[4pt]
  $F[e^{-x^{2}/2}] = \sqrt{2\pi}e^{-u^{2}/2},  F[xe^{-x^{2}/2}] = -i\sqrt{2\pi}ue^{-u^{2}/2}$（問 7(1)）\\[4pt]
  積 $: \sqrt{2\pi}e^{-u^{2}/2}\cdot(-i\sqrt{2\pi}ue^{-u^{2}/2}) = -2\pi iu\cdot e^{-u^{2}}$\\[4pt]
  $(2) F^{-1}[-2\pi iue^{-u^{2}}]$ を計算\\[4pt]
  $F^{-1}[e^{-u^{2}}] = e^{-x^{2}/4}/(2\sqrt{\pi})$（問 8 で $b=1$）\\[4pt]
  $I(x)=\int e^{-u^{2}}e^{iux}du = \sqrt{\pi} e^{-x^{2}/4}$ の微分 $: \int ue^{-u^{2}}e^{iux}du = (ix/2)\sqrt{\pi} e^{-x^{2}/4}$\\[4pt]
  $    F^{-1}[-2\pi iue^{-u^{2}}] = (-i)\cdot(1/2\pi)\cdot2\pi\cdot(ix/2)\sqrt{\pi} e^{-x^{2}/4} = (x/2)\sqrt{\pi}\cdot e^{-x^{2}/4} = (\sqrt{\pi}/2)xe^{-x^{2}/4}$
  \\[8pt]\textcolor{red}{\textbf{答}}：証明終
\end{frame}
