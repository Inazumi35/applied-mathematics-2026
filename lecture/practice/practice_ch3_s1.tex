\noindent\textbf{\large【フーリエ展開】}
\vspace{0.3em}

\mondai{1}{
次の周期 $2$ の関数 $f(x)$ のフーリエ級数を求めよ．\\
$f(x) = \begin{cases} 1 & (-1 \leq x < 0) \\ 1-x & (0 \leq x < 1) \end{cases},\quad f(x+2) = f(x)$
}
\kotae{
$l=1$．
$a_0 = \displaystyle\int_{-1}^1 f(x)\,dx = 1 + \dfrac{1}{2} = \dfrac{3}{2}$．\\[2pt]
$a_n = \displaystyle\int_{-1}^0 \cos n\pi x\,dx + \int_0^1 (1-x)\cos n\pi x\,dx$\\
$\hphantom{a_n} = 0 + \dfrac{1-(-1)^n}{n^2\pi^2}$．\\[2pt]
$b_n = \displaystyle\int_{-1}^0 \sin n\pi x\,dx + \int_0^1 (1-x)\sin n\pi x\,dx$\\
$\hphantom{b_n} = \dfrac{(-1)^n - 1}{n\pi} + \dfrac{1}{n\pi} = \dfrac{(-1)^n}{n\pi}$．
\[
  f(x) = \dfrac{3}{4} + \sum_{n=1}^\infty \left[\dfrac{1-(-1)^n}{n^2\pi^2}\cos n\pi x + \dfrac{(-1)^n}{n\pi}\sin n\pi x\right]
\]
}

\mondai{2}{
グラフが図で示されている次の関数のフーリエ級数を求めよ．\\
(1) 三角波：$f(x) = 2 - |x|\ (-2 \leq x < 2),\ f(x+4) = f(x)$\\
(2) のこぎり波：$f(x) = x\ (-1 \leq x < 1),\ f(x+2) = f(x)$
}
\kotae{
(1) 周期 $4,\ l=2$．偶関数なので $b_n=0$．\\
$a_0 = \displaystyle\int_0^2 (2-x)\,dx = 2$．\\[2pt]
$a_n = \displaystyle\int_0^2 (2-x)\cos\dfrac{n\pi x}{2}\,dx = \dfrac{4(1-(-1)^n)}{n^2\pi^2}$．
\[
  f(x) = 1 + \sum_{k=1}^\infty \dfrac{8}{(2k-1)^2\pi^2}\cos\dfrac{(2k-1)\pi x}{2}
\]
(2) 周期 $2,\ l=1$．奇関数なので $a_n=0$．\\
$b_n = 2\displaystyle\int_0^1 x\sin n\pi x\,dx = -\dfrac{2(-1)^n}{n\pi} = \dfrac{2(-1)^{n+1}}{n\pi}$．
\[
  f(x) = \sum_{n=1}^\infty \dfrac{2(-1)^{n+1}}{n\pi}\sin n\pi x
\]
}

\mondai{3}{
次の関数について以下の問いに答えよ．
$f(x) = x^2\ (-1 \leq x < 1),\ f(x+2) = f(x)$\\
(1) フーリエ級数を求めよ．\\
(2) $x=0$ および $x=1$ とおくことにより
$1 - \dfrac{1}{2^2} + \dfrac{1}{3^2} - \cdots = \dfrac{\pi^2}{12}$，
$1 + \dfrac{1}{2^2} + \dfrac{1}{3^2} + \cdots = \dfrac{\pi^2}{6}$ を導け．
}
\kotae{
(1) 偶関数なので $b_n=0$．
$a_0 = \displaystyle\int_{-1}^1 x^2\,dx = \dfrac{2}{3}$．\\[2pt]
$a_n = 2\displaystyle\int_0^1 x^2 \cos n\pi x\,dx = \dfrac{4(-1)^n}{n^2\pi^2}$（部分積分2回）．
\[
  f(x) = \dfrac{1}{3} + \sum_{n=1}^\infty \dfrac{4(-1)^n}{n^2\pi^2}\cos n\pi x
\]
(2) $x=0$ で $f(0)=0$ より
$0 = \dfrac{1}{3} + \dfrac{4}{\pi^2}\sum (-1)^n/n^2$，すなわち
$\sum_{n=1}^\infty \dfrac{(-1)^{n+1}}{n^2} = \dfrac{\pi^2}{12}$．\\[2pt]
$x=1$ で $f$ は連続で $f(1)=1$ より
$1 = \dfrac{1}{3} + \dfrac{4}{\pi^2}\sum 1/n^2$，すなわち
$\sum_{n=1}^\infty \dfrac{1}{n^2} = \dfrac{\pi^2}{6}$．
}

\mondai{4}{
次の関数 $f(x)$ の複素フーリエ級数を求めよ．\\
$f(x) = e^x\ (-\pi/2 \leq x < \pi/2),\ f(x+\pi) = f(x)$
}
\kotae{
周期 $\pi,\ l=\pi/2$．複素フーリエ係数は
\[
  c_n = \dfrac{1}{\pi}\int_{-\pi/2}^{\pi/2} e^{x} e^{-2inx}\,dx
      = \dfrac{1}{\pi(1-2in)}\bigl[e^{(1-2in)x}\bigr]_{-\pi/2}^{\pi/2}
\]
$e^{(1-2in)\pi/2} = (-1)^n e^{\pi/2},\ e^{-(1-2in)\pi/2} = (-1)^n e^{-\pi/2}$ より
\[
  c_n = \dfrac{(-1)^n (e^{\pi/2}-e^{-\pi/2})}{\pi(1-2in)} = \dfrac{2(-1)^n \sinh(\pi/2)}{\pi(1-2in)}
\]
\[
  f(x) = \dfrac{2\sinh(\pi/2)}{\pi}\sum_{n=-\infty}^\infty \dfrac{(-1)^n}{1-2in} e^{2inx}
\]
}
