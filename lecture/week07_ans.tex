  線形システムの伝達関数とデルタ関数
  \end{block}
  \vfill\hfill{\scriptsize \textcolor{gray}{第2章 ラプラス変換　§2}}

\begin{frame}{線形システム}
  \begin{block}{定数係数線形常微分方程式で記述されるシステム}
  \[
    y'' + ay' + by = x(t), \quad y(0) = 0,\ y'(0) = 0
  \]
  {\footnotesize \textcolor{gray}{入力 $x(t)$、出力 $y(t)$}}
  \end{block}
\end{frame}

\begin{frame}{伝達関数}
  両辺をラプラス変換すると\\[4pt]
  $(s^{2} + as + b)Y(s) = X(s)$\\[8pt]
  \begin{block}{伝達関数}
  \[
    H(s) = \frac{Y(s)}{X(s)} = \frac{1}{s^{2} + as + b}
  \]
  \end{block}
  \vspace{4pt}
  $h(t) = L^{-1}[H(s)]$：インパルス応答\\[4pt]
  $y(t) = h(t) * x(t) = \displaystyle\int_{0}^t h(t-\tau)\, x(\tau)\, d\tau$
\end{frame}

\begin{frame}{演習問題}
  \begin{exampleblock}{自分で解いてみよう}
  \begin{description}
    \item[問10] 微分方程式 $y'' + y' - 2y = x(t), y(0)=0, y'(0)=0$ の
    伝達関数 $H(s)$ を求めよ。また、出力 $y(t)$ を入力 $x(t)$ で表せ\\
  \end{description}
  \end{exampleblock}
\end{frame}

\begin{frame}{補足：部分分数分解（問10）}
  $\dfrac{1}{(s-1)(s+2)} = \dfrac{A}{s-1} + \dfrac{B}{s+2}$ とおく\\[8pt]
  両辺に $(s-1)(s+2)$ を掛けて
  \[
    1 = A(s+2) + B(s-1)
  \]
  $s=1$ を代入：$1 = 3A \;\Rightarrow\; A = \dfrac{1}{3}$\\[6pt]
  $s=-2$ を代入：$1 = -3B \;\Rightarrow\; B = -\dfrac{1}{3}$\\[8pt]
  \[
    \therefore\quad \frac{1}{(s-1)(s+2)} = \frac{1}{3}\cdot\frac{1}{s-1} - \frac{1}{3}\cdot\frac{1}{s+2}
  \]
\end{frame}

\begin{frame}{問10　解答}
  初期値ゼロで両辺をラプラス変換:\\[4pt]
  $(s^{2}+s-2)Y(s) = X(s)$\\[4pt]
  $H(s) = Y(s)/X(s) = 1/(s^{2}+s-2) = 1/((s-1)(s+2))$\\[8pt]
  $h(t) = L^{-1}[H(s)] = L^{-1}\left[\dfrac{1/3}{s-1} - \dfrac{1/3}{s+2}\right] = \dfrac{1}{3}(e^{t} - e^{-2t})$\\[4pt]
  $y(t) = h(t) * x(t) = \dfrac{1}{3}\int_{0}^t x(\tau)(e^{t-\tau} - e^{-2(t-\tau)})\, d\tau$
  \\[8pt]\textcolor{red}{\textbf{答}}：$H(s) = 1/((s-1)(s+2))$、$y(t) = \dfrac{1}{3}\int_{0}^t x(\tau)(e^{t-\tau} - e^{-2(t-\tau)})\, d\tau$
\end{frame}

\begin{frame}{デルタ関数の導入}
  \begin{block}{定義：近似関数}
  \[
    \varphi_\varepsilon(t) = \begin{cases} 1/\varepsilon & (0 \leq t < \varepsilon) \\ 0 & (t < 0 \text{ または } t \geq \varepsilon) \end{cases}
  \]
  {\footnotesize \textcolor{gray}{極めて短時間に働く衝撃的な力やインパルス電圧などを表す}}
  \end{block}
\end{frame}

\begin{frame}{デルタ関数（ディラックのデルタ関数）}
  \begin{block}{定義}
  \[
    \delta(t) = \lim_{\varepsilon \to +0} \varphi_\varepsilon(t)
  \]
  \end{block}
  \vspace{8pt}
  性質：\\[4pt]
  \begin{itemize}
    \item $\delta(t) = 0 \quad (t \neq 0)$
    \item $\displaystyle\int_{0}^{\infty} \delta(t)\, dt = 1$
  \end{itemize}
\end{frame}

\begin{frame}{デルタ関数のラプラス変換}
  \begin{block}{}
  \[
    L[\delta(t)] = 1
  \]
  \end{block}
  \vspace{4pt}
  導出：$L[\varphi_\varepsilon(t)] = \int_0^\varepsilon (1/\varepsilon) e^{-st}\, dt = \dfrac{1-e^{-\varepsilon s}}{\varepsilon s}$\\[8pt]
  $\varepsilon \to +0$ でロピタルの定理より $\to 1$
\end{frame}

\begin{frame}{補足：ロピタルの定理の適用}
  $L[\varphi_\varepsilon(t)] = \dfrac{1-e^{-\varepsilon s}}{\varepsilon s}$ の $\varepsilon\to+0$ の極限を求める。\\[8pt]
  $\varepsilon=0$ を代入すると分子・分母ともに $0$ → $\boldsymbol{0/0}$ 型\\[8pt]
  \begin{block}{ロピタルの定理}
    分子・分母をそれぞれ $\varepsilon$ で微分して極限をとる
  \end{block}
  \vspace{4pt}
  分子 $\dfrac{d}{d\varepsilon}(1-e^{-\varepsilon s}) = se^{-\varepsilon s}$\\[6pt]
  分母 $\dfrac{d}{d\varepsilon}(\varepsilon s) = s$\\[8pt]
  \[
    \lim_{\varepsilon\to+0}\frac{1-e^{-\varepsilon s}}{\varepsilon s}
    = \lim_{\varepsilon\to+0}\frac{se^{-\varepsilon s}}{s}
    = e^{0} = 1
  \]
\end{frame}

\begin{frame}{補足：積分の平均値の定理}
  \begin{block}{定理}
    $h(u)$ が連続なとき、ある $c\in(t-\varepsilon,\, t)$ が存在して
    \[
      \frac{1}{\varepsilon}\int_{t-\varepsilon}^{t} h(u)\,du = h(c)
    \]
  \end{block}
  \vspace{4pt}
  \textbf{直感}：区間 $[t-\varepsilon, t]$ 上の $h$ の「平均値」はどこかの点での値に等しい。\\[8pt]
  $\varepsilon\to+0$ とすると区間が縮まり $c\to t$、連続性より
  \[
    h(c) \to h(t)
  \]
  \textbf{ポイント}：$h$ が連続であることが前提。インパルス応答 $h(t)$ はなめらかな関数なので適用できる。
\end{frame}

\begin{frame}{デルタ入力とインパルス応答}
  入力 $\varphi_\varepsilon(t)$ に対する出力：\\[4pt]
  $y_\varepsilon(t) = \int_0^t \varphi_\varepsilon(\tau)\, h(t-\tau)\, d\tau = (1/\varepsilon)\int_0^\varepsilon h(t-\tau)\, d\tau \quad (t > \varepsilon)$\\[4pt]
  $u = t-\tau$ と置換：$= (1/\varepsilon)\int_{t-\varepsilon}^t h(u)\, du$\\[4pt]
  積分の平均値の定理より $= h(c) \quad (t-\varepsilon < c < t)$\\[4pt]
  $\varepsilon \to +0$ とすると：$\lim y_\varepsilon(t) = h(t)$\\[8pt]
  \begin{alertblock}{}
  $\delta(t)$ 入力に対する出力はインパルス応答 $h(t)$ そのものである
  \end{alertblock}
\end{frame}

\begin{frame}{演習問題}
  \begin{exampleblock}{自分で解いてみよう}
  \begin{description}
    \item[問12] 次の微分方程式の解を求めよ。ただし $\omega$ は正の定数とする。
    $y'' + \omega^{2}y = \delta(t),  y(0) = 0,  y'(0) = 0$\\
  \end{description}
  \end{exampleblock}
\end{frame}

\begin{frame}{補足：$\delta(\tau)$ が $\tau=0$ を「選び出す」理由}
  デルタ関数の性質：
  \begin{itemize}
    \item $\delta(\tau) = 0 \quad (\tau \neq 0)$
    \item $\displaystyle\int_{0}^{\infty} \delta(\tau)\,d\tau = 1$
  \end{itemize}
  \vspace{6pt}
  これを使うと $\tau\neq 0$ では $\delta(\tau)=0$ なので被積分関数は $\tau=0$ の近傍だけに寄与する。\\[6pt]
  $f(t-\tau)$ は $\tau=0$ の近傍では $f(t-0)=f(t)$ とみなせるから
  \[
    \int_{0}^{t} \delta(\tau)\,f(t-\tau)\,d\tau
    = f(t)\int_{0}^{t} \delta(\tau)\,d\tau = f(t)\cdot 1 = f(t)
  \]
  \textbf{まとめ}：$\delta(\tau)$ との積分は $\tau=0$ の値だけを取り出す。
\end{frame}

\begin{frame}{問11　解答}
  ラプラス変換を使った証明:\\[4pt]
  $L[f(t)*\delta(t)] = F(s)\cdot L[\delta(t)] = F(s)\cdot1 = F(s) = L[f(t)]$\\[4pt]
  $\therefore f(t)*\delta(t) = f(t)$\\[4pt]
  交換法則より $\delta(t)*f(t) = f(t)*\delta(t) = f(t)$ （証明終）\\[4pt]
  （直接証明） $\delta(t)*f(t) = \int_{0}^t \delta(\tau)f(t-\tau)\, d\tau = f(t-0) = f(t)$
\end{frame}

\begin{frame}{問12　解答}
  両辺をラプラス変換（初期値ゼロ）:\\[4pt]
  $(s^{2} + \omega^{2})Y(s) = L[\delta(t)] = 1$\\[4pt]
  $Y(s) = 1/(s^{2}+\omega^{2}) = (1/\omega)\cdot\omega/(s^{2}+\omega^{2})$\\[4pt]
  逆ラプラス変換 $: y(t) = (1/\omega)\sin(\omega t)$
  \\[8pt]\textcolor{red}{\textbf{答}}：$y(t) = (1/\omega)\sin(\omega t)$（デルタ入力に対するインパルス応答）
\end{frame}
